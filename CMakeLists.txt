cmake_minimum_required(VERSION 3.24)
project(CoopSbR VERSION 0.1.0 LANGUAGES CXX)
option(COOPSBR_ENABLE_CUDA "Build the CUDA implementations" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(coopsbr-core src/io.cpp src/sequential.cpp)
target_include_directories(coopsbr-core PUBLIC include)
target_compile_options(coopsbr-core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
add_executable(coopsbr src/main.cpp)
target_link_libraries(coopsbr PRIVATE coopsbr-core)
add_executable(coopsbr-check src/check.cpp)
target_link_libraries(coopsbr-check PRIVATE coopsbr-core)
if(COOPSBR_ENABLE_CUDA)
  set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
  enable_language(CUDA)
  add_library(coopsbr-cuda STATIC src/gpu.cu)
  target_link_libraries(coopsbr-cuda PUBLIC coopsbr-core)
  target_compile_definitions(coopsbr-cuda PUBLIC COOPSBR_HAS_CUDA=1)
  set_target_properties(coopsbr-cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES
    CUDA_ARCHITECTURES 89-real CUDA_SEPARABLE_COMPILATION ON CUDA_RESOLVE_DEVICE_SYMBOLS ON)
  target_link_libraries(coopsbr PRIVATE coopsbr-cuda)
  add_executable(coopsbr-update-bench src/update_bench.cpp)
  target_link_libraries(coopsbr-update-bench PRIVATE coopsbr-cuda)
  add_executable(coopsbr-gpu-test tests/gpu_test.cpp)
  target_link_libraries(coopsbr-gpu-test PRIVATE coopsbr-cuda)
endif()
enable_testing()
add_executable(coopsbr-policy-test tests/policy_test.cpp)
target_link_libraries(coopsbr-policy-test PRIVATE coopsbr-core)
add_test(NAME exact-policy COMMAND coopsbr-policy-test 8)

if(COOPSBR_ENABLE_CUDA)
  add_test(NAME gpu-equivalence COMMAND coopsbr-gpu-test 5)
  set_tests_properties(gpu-equivalence PROPERTIES TIMEOUT 180)
endif()

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME workflow COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_workflow.py
    --binary $<TARGET_FILE:coopsbr> --checker $<TARGET_FILE:coopsbr-check>)
  set_tests_properties(workflow PROPERTIES TIMEOUT 60)
endif()

if(Python3_Interpreter_FOUND)
  add_test(NAME real-corpus-integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/validate_corpus.py
    --manifest ${CMAKE_SOURCE_DIR}/datasets/real/v1/manifest.json)
  add_test(NAME synthetic-corpus-integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/validate_corpus.py
    --manifest ${CMAKE_SOURCE_DIR}/datasets/synthetic/v1/manifest.json)
  set_tests_properties(real-corpus-integrity synthetic-corpus-integrity PROPERTIES TIMEOUT 60)
endif()
